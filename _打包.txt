"""表格体检：双击运行，按总纲规则检查 text.csv 和 order.csv，有问题说人话。
v0.2：总页数以 manifest.json 为准（不再拿 text.csv 行数推）；
      text.csv 允许同页多行（格子填字一页多格），no 只校验范围。
v0.3：manifest 读取提前到函数开头，与 text.csv 是否存在解耦；
      修掉"text.csv 缺失/为空时误报'没有 manifest.json'"的文案谎言。
"""
import csv
import json
import sys
from pathlib import Path

def read_rows(path):
    with open(path, encoding="utf-8-sig") as f:
        return list(csv.reader(f))

def _page_total(work_dir):
    """总页数以 manifest.json 为准，不再用 text.csv 行数推。"""
    try:
        data = json.load(open(work_dir / "manifest.json", encoding="utf-8"))
        pages = data.get("pages", data)
        return len(pages)
    except Exception:
        return None

def check(work_dir, errors, warns):
    # 总页数提前读，与 text.csv 是否存在无关
    total = _page_total(work_dir)
    if total is None:
        errors.append("找不到/读不了 manifest.json，页数校验跳过——先重新跑第一步拆页")

    # ---- 检查 text.csv ----
    tpath = work_dir / "text.csv"
    if not tpath.exists():
        errors.append(f"找不到 {tpath}，先跑第一步拆页")
    else:
        rows = read_rows(tpath)
        if not rows:
            errors.append("text.csv 是空的，重新跑第一步拆页")
        else:
            header, body = rows[0], rows[1:]
            if len(header) != 8:
                errors.append(f"text.csv 表头应是 8 格，现在 {len(header)} 格：{header}")
            for i, r in enumerate(body, start=2):
                if len(r) != 8:
                    errors.append(f"text.csv 第 {i} 行格子数是 {len(r)}，应是 8 格（英文逗号分隔）")
                    continue
                no = r[0].strip()
                if not no:
                    errors.append(f"text.csv 第 {i} 行 no 列是空的")
                    continue
                if not no.isdigit():
                    errors.append(f"text.csv 第 {i} 行 no 不是数字：{no}")
                    continue
                if total and not 1 <= int(no) <= total:
                    errors.append(f"text.csv 第 {i} 行 no={no} 超出页数范围（共 {total} 页）")
            warns.append(f"text.csv 共 {len(body)} 行（同页可多行），"
                         f"填了字的 {sum(1 for r in body if len(r)>1 and r[1].strip())} 行")

    # ---- 检查 order.csv ----
    opath = work_dir / "order.csv"
    if not opath.exists():
        errors.append(f"找不到 {opath}，先跑第一步拆页")
    else:
        rows = read_rows(opath)
        if not rows:
            errors.append("order.csv 是空的，重新跑第一步拆页")
        else:
            header, body = rows[0], rows[1:]
            if header[:3] != ["new_no", "old_no", "replace_from"]:
                errors.append(f"order.csv 表头应是 new_no,old_no,replace_from，现在是：{header}")
            news, olds = [], []
            for i, r in enumerate(body, start=2):
                try:
                    new_no = int((r[0] if len(r) > 0 else "").strip())
                    old_no = int((r[1] if len(r) > 1 else "").strip())
                except ValueError:
                    errors.append(f"order.csv 第 {i} 行页码不是数字：{r[:2]}")
                    continue
                news.append(new_no)
                olds.append(old_no)
                if len(r) > 2 and r[2].strip():
                    try:
                        rf = int(r[2].strip())
                        if total and not 1 <= rf <= total:
                            errors.append(f"order.csv 第 {i} 行 replace_from={rf} 超出范围（共 {total} 页）")
                    except ValueError:
                        errors.append(f"order.csv 第 {i} 行 replace_from 不是数字：{r[2]}")
            dup = {n for n in news if news.count(n) > 1}
            if dup:
                errors.append(f"order.csv 的 new_no 有重复：{sorted(dup)}——不许加行删行")
            m_rows = len(body)
            if total:
                miss = [n for n in range(1, m_rows + 1) if n not in news]
                if miss:
                    errors.append(f"order.csv 缺了这些 new_no：{miss}——位置号必须从1连续")
                if m_rows > total:
                    errors.append(f"order.csv 有 {m_rows} 行，比总页数 {total} 还多——删页用口令，别手工加行")
            blanks = 0
            for i, n in enumerate(olds, start=2):
                if n == 0:
                    blanks += 1
                    continue
                if total and not 1 <= n <= total:
                    errors.append(f"order.csv 第 {i} 行 old_no={n} 超出范围（共 {total} 页）")
            d2 = {n for n in olds if n != 0 and olds.count(n) > 1}
            if d2:
                errors.append(f"order.csv 的 old_no 有重复：{sorted(d2)}——每页内容只能被用一个位置（0=空白页不算）")
            if blanks:
                warns.append(f"order.csv 有 {blanks} 张空白页")

            moved = sum(1 for r in body if len(r) > 1 and r[0].strip() != r[1].strip())
            warns.append(f"order.csv 换了 {moved} 页位置，强行换图 {sum(1 for r in body if len(r)>2 and r[2].strip())} 页")

def main():

"""共用底层：配置加载、输出目录、日志、异常分级。"""
import json
import hashlib
import shutil
from pathlib import Path
from datetime import datetime

PROJECT_ROOT = Path(__file__).resolve().parent.parent

class FatalError(Exception):
    """致命错误：输出目录不可写、输入损坏、依赖缺失——立即停止。"""
    pass

def load_config():
    with open(PROJECT_ROOT / "config.json", encoding="utf-8") as f:
        return json.load(f)

def file_hash(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()

def now_str():
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")

def ensure_output_dir(user_dir=None, source_file=None, task_name="任务"):
    """输出目录三分：最终文件 / 工作文件 / 日志。返回三个 Path。"""
    if user_dir:
        root = Path(user_dir)
    else:
        cfg = load_config()
        if cfg["output_policy"] == "source_dir" and source_file:
            date = datetime.now().strftime("%Y%m%d")
            root = Path(source_file).parent / f"输出_{date}" / task_name
        else:
            root = PROJECT_ROOT / "输出" / task_name

    # 按源 PDF 隔离：每个输入文件有独立的工作/最终/日志目录，杜绝混用
    if source_file:
        root = root / Path(source_file).stem

    try:
        root.mkdir(parents=True, exist_ok=True)
        test = root / ".write_test"
        test.touch(); test.unlink()
    except OSError as e:
        raise FatalError(f"输出目录不可写: {root} ({e})")

    final_dir = root / "最终文件"
    work_dir  = root / "工作文件"
    log_dir   = root / "日志"
    for d in (final_dir, work_dir, log_dir):
        d.mkdir(exist_ok=True)
    return final_dir, work_dir, log_dir

def unique_path(path):
    """冲突策略：自动加序号，永不覆盖。"""
    path = Path(path)
    if not path.exists():
        return path
    i = 1
    while True:
        cand = path.with_name(f"{path.stem}_{i}{path.suffix}")
        if not cand.exists():
            return cand
        i += 1

class Logger:
    """每步一行、带时间戳。日志是记事本环境下唯一的调试手段。"""
    def __init__(self, log_dir, name="run"):
        self.path = log_dir / f"{name}.log"
        self._write("=== 运行开始 ===")

    def _write(self, msg):
        line = f"[{now_str()}] {msg}"
        print(line)
        with open(self.path, "a", encoding="utf-8") as f:
            f.write(line + "\n")

    def info(self, msg):  self._write(msg)
    def warn(self, msg):  self._write(f"[警告] {msg}")
    def error(self, msg): self._write(f"[错误] {msg}")

# -*- coding: utf-8 -*-
"""按编号插入文字。只填 no,text 时用默认规则；高级用户可填坐标和样式。"""
import csv
import fitz
from pathlib import Path
from .common import load_config

def load_text_csv(work_dir):
    rows = []
    path = work_dir / "text.csv"
    with open(path, encoding="utf-8-sig", newline="") as f:
        for row in csv.DictReader(f):
            rows.append(row)
    nos = [r["no"] for r in rows]
    if len(nos) != len(set(nos)):
        raise ValueError("text.csv 中 no 列有重复，请修正后再运行")
    return rows

def get_font():
    """优先 config 指定字体文件，否则用内置中文字体。"""
    cfg = load_config()
    fp = cfg.get("font_path", "")
    if fp:
        from pathlib import Path as P
        if P(fp).exists():
            return "custom", fp
    return "builtin", cfg.get("font_fallback", "china-s")

def insert_text(input_pdf, output_pdf, work_dir, logger):
    rows = load_text_csv(work_dir)
    font_kind, font_val = get_font()
    doc = fitz.open(input_pdf)
    inserted, missing = 0, []

    for r in rows:
        text = (r.get("text") or "").strip()
        if not text:
            if r.get("status") in ("待审核", "缺失"):
                missing.append({"no": r["no"], "reason": "缺文字",
                                "note": "text 为空"})
            continue
        try:
            page = doc[int(r["no"]) - 1]  # 编号即页序
            x = float(r["x"]) if r.get("x") else 72
            y = float(r["y"]) if r.get("y") else 72
            size = float(r["size"]) if r.get("size") else 12
            if font_kind == "custom":
                page.insert_font(fontname="userfont", fontfile=font_val)
                fname = "userfont"
            else:
                fname = font_val
            page.insert_text((x, y), text, fontname=fname, fontsize=size)
            r["status"] = "已确认"
            inserted += 1
            logger.info(f"页 {r['no']} 插字: {text[:20]}")
        except Exception as e:
            missing.append({"no": r["no"], "reason": "坐标无效或页面损坏",
                            "note": str(e)})
            logger.warn(f"页 {r['no']} 插字失败: {e}")

    doc.save(str(output_pdf), deflate=True)
    doc.close()

    if missing:
        with open(work_dir / "missing.csv", "w", encoding="utf-8-sig",
                  newline="") as f:
            w = csv.DictWriter(f, fieldnames=["no", "reason", "note"])
            w.writeheader()
            w.writerows(missing)
        logger.warn(f"有 {len(missing)} 条缺失，见 missing.csv")
    logger.info(f"插字完成，共插入 {inserted} 条")
    return missing

"""order.csv：自由重排 + 替换。内容跟随来源页，位置跟随 new_no。"""
import csv
from .common import FatalError

def _read_dicts(path):
    with open(path, encoding="utf-8-sig") as f:
        return list(csv.DictReader(f))

def load_order(work_dir, total, logger):
    """返回 {新页码: 内容页码}。无表 = 原序 1:1。"""
    path = work_dir / "order.csv"
    if not path.exists():
        logger.info("未找到 order.csv，按原页序重建")
        return {i: i for i in range(1, total + 1)}

    mapping = {}
    old_nos = []
    for r in _read_dicts(path):
        try:
            new_no = int((r.get("new_no") or "").strip())
            old_no = int((r.get("old_no") or "").strip())
        except ValueError:
            raise FatalError(f"order.csv 中页码不是数字: {dict(r)}")
        if old_no != 0:
            old_nos.append(old_no)
        rf = (r.get("replace_from") or "").strip()
        content = int(rf) if rf else old_no
        # 位置号必须在 1..total；内容号 0 = 空白页，合法
        if not 1 <= new_no <= total:
            raise FatalError(f"order.csv 里 new_no={new_no} 超出范围（共 {total} 页）")
        if rf and not 1 <= int(rf) <= total:
            raise FatalError(f"order.csv 里 replace_from={rf} 超出范围（共 {total} 页）")
        if content != 0 and not 1 <= content <= total:
            raise FatalError(f"order.csv 里 old_no={old_no} 超出范围（共 {total} 页）")
        if new_no in mapping:
            raise FatalError(f"order.csv 中 new_no={new_no} 重复，每行只许出现一次")
        mapping[new_no] = content
    # 位置号必须从 1 开始连续（删页后行数变少也必须是 1..M）
    if sorted(mapping) != list(range(1, len(mapping) + 1)):
        raise FatalError("order.csv 的 new_no 必须从 1 开始连续编号（不许跳号、不许重复）")
    # old_no 除 0（空白）外不许重复；replace_from 产生的
    # 同内容多引用（复制）是允许的，不在此列
    if len(old_nos) != len(set(old_nos)):
        raise FatalError("order.csv 的 old_no 有重复（除空白页外，每页内容只能用一个位置）")
    for new_no, content in mapping.items():
        if content != new_no:
            logger.info(f"页位 {new_no} ← 内容来自原第 {content} 页" if content
                        else f"页位 {new_no} ← 空白页")
    return mapping

"""按 order.csv 重排/替换页面，并按 text.csv 插字、sticker.csv 贴图，一步重建 PDF。"""
import fitz
import csv
import json
from pathlib import Path

COLORS = {"black": (0, 0, 0), "red": (1, 0, 0), "blue": (0, 0, 1),
          "green": (0, 0.5, 0), "gray": (0.5, 0.5, 0.5)}

def _parse_color(s):
    """认名字也认 hex（#ff0000 / #f00），认不出回退黑色。"""
    s = (s or "").strip().lower()
    if s in COLORS:
        return COLORS[s]
    if s.startswith("#"):
        h = s.lstrip("#")
        if len(h) == 3:
            h = "".join(c * 2 for c in h)
        if len(h) == 6:
            return tuple(int(h[i:i + 2], 16) / 255 for i in (0, 2, 4))
    return (0, 0, 0)

def _read_dicts(path):
    with open(path, encoding="utf-8-sig") as f:
        return list(csv.DictReader(f))

def _insert_text(page, row, logger):
    text = (row.get("text") or "").strip()
    if not text:
        return
    def num(key, default):
        v = (row.get(key) or "").strip()
        return float(v) if v else default
    size = num("size", 14)
    x, y = num("x", 50), num("y", 50)
    color = _parse_color(row.get("color"))
    align = (row.get("align") or "").strip()
    if align in ("中", "居中", "center", "右", "右对齐", "right"):
        try:
            w = fitz.get_text_length(text, fontname="china-s", fontsize=size)
        except Exception:
            w = len(text) * size
        if align.startswith(("中", "居中", "c")):
            x = page.rect.width / 2 - w / 2
        else:
            x = page.rect.width - x - w
    page.insert_text((x, y), text, fontname="china-s", fontsize=size, color=color)

def _load_stickers(work_dir):

        logger.info(f"页 {page_no}: 已贴 {len(mine)} 张图")
    except Exception as e:
        logger.warn(f"页 {page_no}: 贴图失败，已跳过（{e}）")

def rebuild(work_dir, output_pdf, logger):
    with open(work_dir / "manifest.json", encoding="utf-8") as f:
        manifest = json.load(f)
    pages = {p["page"]: p for p in manifest["pages"]}
    total = len(manifest["pages"])

    mapping = load_order(work_dir, total, logger)   # 新页码 -> 内容页码

    texts = {}
    tpath = work_dir / "text.csv"
    if tpath.exists():
        for r in _read_dicts(tpath):
            no = (r.get("no") or "").strip()
            if no:
                texts.setdefault(no, []).append(r)   # 同页多行都保留

    stk_rows = _load_stickers(work_dir)              # 贴图一次读入，循环里查

    # 空白页尺寸：跟随原 PDF（用第一张正常页的尺寸当参照）
    ref = next((p for p in manifest["pages"] if p.get("status") == "已确认"), None)
    blank_w = ref["width_pt"] if ref else 595
    blank_h = ref["height_pt"] if ref else 842

    doc = fitz.open()

    for new_no in sorted(mapping):
        if mapping[new_no] == 0:                          # 空白页
            doc.new_page(width=blank_w, height=blank_h)
            logger.info(f"新页 {new_no}: 空白页")
            continue
        src = pages[mapping[new_no]]

        if src.get("status") == "缺失":
            page = doc.new_page(width=blank_w, height=blank_h)
            page.insert_text((72, 100), f"[占位页 {new_no}]",
                             fontname="china-s", fontsize=14)
            logger.warn(f"新页 {new_no}（原页 {src['page']}）缺失，用占位页代替")
            continue
        page = doc.new_page(width=src["width_pt"], height=src["height_pt"])
        page.insert_image(page.rect, filename=str(work_dir / src["image"]))
        # 字跟随内容页：用内容页的 no 去 text.csv 里找（同页可有多行，逐行插入）
        for row in texts.get(src["no"], []):
            _insert_text(page, row, logger)
        # 贴图：必须盖在文字之后，否则图会压住字
        _paste_stickers(page, stk_rows, src["no"], logger)
    doc.save(str(output_pdf), deflate=True)
    doc.close()
    logger.info(f"重建完成（含重排/替换/插字/贴图）: {output_pdf}")

# -*- coding: utf-8 -*-
"""拆页编号：输入只读，渲染成单页 PNG，生成 manifest.json、text.csv、order.csv。"""
import fitz  # PyMuPDF
import csv
import json
from pathlib import Path
from .common import file_hash, load_config, now_str, FatalError

def split_pdf(input_path, work_dir, logger):

    cfg = load_config()
    dpi = cfg["render_dpi"]
    zoom = dpi / 72.0  # PDF 点 -> 像素
    doc = fitz.open(input_path)  # 只读打开，从不写回

    pages = []
    for i, page in enumerate(doc):
        no = f"{i+1:03d}"  # 编号靠页序，不靠 AI 猜
        img_name = f"{no}.png"
        try:
            pix = page.get_pixmap(matrix=fitz.Matrix(zoom, zoom))
            pix.save(work_dir / img_name)
            pages.append({
                "no": no,
                "source": str(input_path),
                "page": i + 1,
                "type": "页面图像",
                "width_pt": round(page.rect.width, 2),   # 坐标统一用 PDF 点
                "height_pt": round(page.rect.height, 2),
                "rotation": page.rotation,
                "dpi": dpi,
                "image": img_name,
                "status": "已确认",
            })
            logger.info(f"页 {no} 拆出: {pix.width}x{pix.height}px")
        except Exception as e:
            # 单页失败：跳过并记录，不停机
            pages.append({"no": no, "page": i + 1, "status": "缺失",
                          "reason": f"页面损坏: {e}", "image": img_name})
            logger.warn(f"页 {no} 拆页失败，已记入缺失: {e}")
    doc.close()

    manifest = {
        "schema_version": load_config()["manifest_schema_version"],
        "script_version": load_config()["version"],
        "input_file": str(input_path),
        "input_sha256": file_hash(input_path),
        "created": now_str(),
        "params": {"dpi": dpi},
        "pages": pages,
    }
    with open(work_dir / "manifest.json", "w", encoding="utf-8") as f:
        json.dump(manifest, f, ensure_ascii=False, indent=2)

    # text.csv：UTF-8-SIG，用户只需填 no 和 text；已存在则不覆盖，保住用户填写的内容
    csv_path = work_dir / "text.csv"
    if not csv_path.exists():
        with open(csv_path, "w", encoding="utf-8-sig", newline="") as f:
            w = csv.writer(f)
            w.writerow(["no", "text", "x", "y", "size", "color", "align", "status"])
            for p in pages:
                if p.get("status") != "缺失":
                    w.writerow([p["no"], "", "", "", "", "", "", "待审核"])

    # order.csv：默认 1→1 原序；已存在则不覆盖，保住用户改过的页序
    order_path = work_dir / "order.csv"
    if not order_path.exists():
        with open(order_path, "w", encoding="utf-8-sig", newline="") as f:
            w = csv.writer(f)
            w.writerow(["new_no", "old_no", "replace_from"])
            for p in pages:
                w.writerow([f"{p['page']:03d}", f"{p['page']:03d}", ""])

    # 原始快照：恢复口令的依据（拆页时的最原始页序，之后永不改动）
    # 注意：与 order.csv 是否存在解耦——只要 origin.csv 缺，就按本次拆页
    # 重建（用于补建老任务缺失的 origin.csv）
    origin_path = work_dir / "origin.csv"
    if not origin_path.exists():
        with open(origin_path, "w", encoding="utf-8-sig", newline="") as f:
            w = csv.writer(f)
            w.writerow(["new_no", "old_no", "replace_from"])
            for p in pages:
                w.writerow([f"{p['page']:03d}", f"{p['page']:03d}", ""])

    logger.info(f"拆页完成，共 {len(pages)} 页，已生成 manifest.json / text.csv / order.csv / origin.csv")

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  PDF 工具箱 使用说明（v0.7.4）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

一、它会做什么

  把一份 PDF 拆成单页图 → 你可以调整页序、自动找到表格空位、
  往格子里填字或往页面任意位置插字 → 最后合成一份新的 PDF。
  你的原始文件【永远不会被改动】，成品另外生成。
  中途关机、手滑都不怕——每一步都自动存盘。


二、四步出成品

  ① 选择文件     点【选择文件】选你的 PDF（首次会自动拆页出表、扫描空位）
  ② 攒指令       换页口令、批量填字、格子填字，都可以先【加入队列】
  ③ 预览确认     点【点图定位置】在页面图上直接点出文字位置，
                 蓝虚线框就是程序找到的表格空位
  ④ 出成品       检查队列无误 → 点【逐条执行并出成品】
                 （跑完自动体检，通过才生成，并核对页数）


三、换页

  1. 单条口令：输进"口令"框 →【立即执行】马上生效，或【加入队列】攒着
       2换3     第2位和第3位的页对调
       2替5     第2位改用原第5页的内容（相当于把5页的图搬来）
       取消2    取消第2位的替换，恢复用自己原页
       看       查看当前页序
       删3      删掉第3位（后面顶上来，成品变薄）
       删2-4    范围删除第2到4位
       空3      第3位变空白页（页数不变）
       恢复     整本回到原始页序（删掉的页也回来）

  2. 批量口令：大框里每行一条，如
       1换2
       3替5
     点【批量立即执行】或【批量加入队列】。
     也可以把《AI口令模板.txt》整份发给 AI（豆包/ChatGLM 都行），
     用大白话说需求，AI 回你口令，逐行粘进来即可。

  3. 整本倒序：点【全部倒序→队列】
     注意：倒序会清掉"替"的标记，需要的话替完再倒，或倒完重新替。


四、插字

  插字分两种玩法，批量框会自动识别：

  ── 玩法一：格子填字（推荐，表格页专用）──

    程序拆页时会自动扫描每页的表格空位（存进 slots.json）。
    口令格式：（页号）R行C列=文字，可连写：

      （1）R2C2=某某项目（1）R4C2=张三（7）R2C2=某某公司

    · R2C2 = 第2行第2列的格子（编号在预览图的蓝框角标上能看到）
    · 同一页可以填多个格子（这是允许的，表会自动加行）
    · 同一个格子重复填 = 覆盖，不会涨行，放心改
    · 位置、字号自动按空位算好，不用手动定坐标
    · 填完点图核对：点【点图定位置】能看到每个格子的位置效果

  ── 玩法二：整页插字（表格外的任意位置）──

    口令格式：（页号）文字，可连写：

      （1）封面标题（2）第一章

    只填文字不带坐标，坐标需要自己定：
    左边选页 → 填【文字】→ 点【点图定位置】→ 在图上点一下
    → x/y 自动填好 → 关窗 → 点【保存此页】。

  小贴士：
    · 点图定位置时，点在蓝虚线框内 = 自动吸附到格子（推荐）
    · 红点只是位置预览，最终字体样式以成品 PDF 为准
    · 字号、颜色都有下拉可选，也能手输（如 10.5、#336699）
    · 坐标单位是 PDF 点（A4 页宽595、高842）


五、指令队列（推荐的干活方式）

  · 攒：所有口令、倒序、批量填字都能【加入队列】，此时什么都没改
  · 审：队列列表里逐条预览，【删除选中】去掉不要的，【清空队列】重来
  · 跑：点【逐条执行并出成品】——
        成功一条就从队列删一条，边跑边存盘；
        中途出错就停下，队列里剩下的就是"还没执行的"；
        修好问题再点一次，只会跑剩余的，已生效的绝不会重复执行；
        全部跑完自动体检两张表，通过才生成成品并核对页数。

  一句话：队列就是你这次任务的"草稿单"，确认了才允许落地。


六、表格体检（自动的，不用你管）

  每次出成品前程序会自动检查两张表，说人话报问题：
    · 格子数不对、页号不是数字 → 多半是 Excel 编辑时格式坏了
    · 页号超出总页数 → 页号写错了
    · order.csv 缺行/多行/页号重复 → 不许手工加行删行
  总页数以工作文件里的 manifest.json 为准（拆页时自动生成），
  text.csv 允许一页多行（同页多格填字就是靠它）。


七、常见问题

  Q：点了【点图定位置】没反应？
  A：需要 Pillow 库支持：pip install pillow，装好重开程序即可。

  Q：预览图里点了位置，但忘了保存？
  A：关窗后记得点【保存此页】；切页也会自动保存，两种都算数。

  Q：格子口令提示"没这格"？
  A：页号或 R行C列 不在 slots.json 里。先【拆页出表】重新扫描；
     如果那页的表格是图片里手画的、线太淡，可能扫不出来，
     退回玩法二手动定坐标。

  Q：日志提示"正在执行任务，请等它跑完再操作"？
  A：程序正在跑队列或拆页，这是保护机制，防止同时改数据出错，
     等日志出现"执行完毕"再继续。

  Q：口令输进去说"没这页"？
  A：页号超出总页数。先点【拆页出表】看总共几页。

  Q：成品里字没出现？
  A：整页插字多半是没定坐标——看日志里的【待办】清单，逐页点图补上。

  Q：执行到一半程序关了会丢吗？
  A：不会。每条指令跑完立即存盘；重新打开选同一个 PDF，
     之前的页序和文字都在。队列里没跑完的条目需要重新加入。

  Q：想推倒重来？
  A：口令【恢复】回到原始页序；【清空此页】清掉某页的字；
     【清空队列】放弃所有攒着的指令；
     彻底重来：关掉程序 → 删掉工作文件里的 text.csv 和 order.csv
     → 重开点【拆页出表】。

  Q：界面能换深色吗？
  A：能。改 gui.py 顶部 THEME = "bootstrap-dark"，保存重开。


八、文件都在哪

  工作文件（两张表、每页的图、空位表 slots.json、页信息 manifest.json）、
  成品 PDF、运行日志，都在源 PDF 旁边的"重建插字"文件夹里，
  点【打开工作文件】直接打开。

  核心文件：
    order.csv   —— 页序（换页口令改的就是它）
    text.csv    —— 要插的文字和位置（插字区改的就是它）
    slots.json  —— 每页的表格空位（格子口令靠它定位）
    manifest.json —— 每页宽高/状态（体检和坐标换算靠它）
  平时不用碰它们，界面上的一切操作都会自动同步进去。

━━━ 照片件插字（纯图片 PDF）━━━

照片扫出来的 PDF 没有文字层，扫格子认不出表格，按下面的路子走：

一、只想往页面某处插字（最常用）
  1. 批量填字框敲：（1）标题（2）第一章 → 点【批量立即填】
  2. 没定过坐标的字会自动落在页面左下角（x=50，底边上方一点），
     日志会提示默认位置
  3. 点【点图定位置】微调：点哪字落哪；保存此页
  → 不用管格子，直接能插

二、页面里有带框线的表格，想往格子里填
  1. 先【拆页出表】（只走文字扫描，照片页扫不出是正常的）
  2. 点【扫图片格子】（就是认照片里的表格框线）
     → 日志逐页报告扫出几个空位；没框线的表格扫不出，属正常
  3. 格子口令：（页号）R行C列=文字，如（3）R2C2=某某公司
  4. 也可以点图定位置，点在蓝虚线框里 = 自动吸附到格子

  说明：
  · 扫图片格子只往没空位的页里填，不会动已扫出的结果
  · 想重新扫全部：删掉工作目录里的 slots.json 再点
  · 没装 numpy 会很慢（约10秒/页），建议先 pip install numpy

- build.bat 若用命令行参数打包，每次会重新生成 spec，
  手改 spec 无效；要改 spec 就让 build.bat 直接用 spec 打包
- 打包入口文件（launcher.py）必须单独测试：
  python launcher.py 能弹出界面，才能打包
  （gui.py 能跑 ≠ launcher 能跑）
- gui.run_gui() 必须写在 try/except 外面，
  放进 except 里会导致 import 成功就静默退出、无界面无报错
- build.bat 要检查 errorlevel，否则打包失败也喊"打包完成"
- 调试期 spec 里 console=True，能看报错；正式发布改回 False
- launcher 已带崩溃日志兜底：exe 启动失败看旁边的 崩溃日志.txt

- bat 里有 chcp 65001 → 必须存 UTF-8；没有 → 必须存 ANSI

【四、CSV 铁律】

【六、批量填字自动落位】
没坐标的文字自动落页面左下角（x=50, y=页高-36），日志提示，

class ScrollFrame(ttkb.Frame):
    def __init__(self, master, **kw):
        super().__init__(master, **kw)
        canvas = tk.Canvas(self, highlightthickness=0)
        self.canvas = canvas
        vsb = ttkb.Scrollbar(self, orient="vertical", command=canvas.yview)
        hsb = ttkb.Scrollbar(self, orient="horizontal", command=canvas.xview)
        self.inner = ttkb.Frame(canvas)
        self.inner.bind("<Configure>",
            lambda e: canvas.configure(scrollregion=canvas.bbox("all")))
        self._win = canvas.create_window((0, 0), window=self.inner, anchor="nw")
        # 窗口变宽时，内容跟着拉宽；窄于 900 时出现横向滚动条
        canvas.bind("<Configure>", self._on_resize)
        canvas.configure(yscrollcommand=vsb.set, xscrollcommand=hsb.set)
        canvas.grid(row=0, column=0, sticky="nsew")
        vsb.grid(row=0, column=1, sticky="ns")
        hsb.grid(row=1, column=0, sticky="ew")
        self.rowconfigure(0, weight=1)
        self.columnconfigure(0, weight=1)
        _wheel_state = {"acc": 0, "job": None}

        def _on_wheel(e):
            _wheel_state["acc"] += e.delta
            if _wheel_state["job"]:
                return
            def _flush():
                _wheel_state["job"] = None
                d = _wheel_state["acc"]
                _wheel_state["acc"] = 0
                canvas.yview_scroll(-int(d / 120), "units")
            _wheel_state["job"] = canvas.after(30, _flush)

        canvas.bind_all("<MouseWheel>", _on_wheel)

    def _on_resize(self, e):
        e.widget.itemconfigure(self._win, width=max(e.width, 900))

    def _listbox(self, parent, **kw):
        """Listbox 是经典 tk 控件（ttkbootstrap 2.x 不提供主题版），
        从当前主题取色手工上色，观感与其他控件统一。"""
        c = self.style.colors
        kw.setdefault("relief", "flat")
        kw.setdefault("borderwidth", 0)
        kw.setdefault("highlightthickness", 1)
        kw.setdefault("activestyle", "none")
        kw.setdefault("font", FONT)
        try:
            kw.setdefault("bg", c.inputbg)
            kw.setdefault("fg", c.inputfg)
            kw.setdefault("selectbackground", c.primary)
            kw.setdefault("selectforeground", c.selectfg)
            kw.setdefault("highlightbackground", c.border)
            kw.setdefault("highlightcolor", c.primary)
        except Exception:
            pass
        return tk.Listbox(parent, **kw)

    def _scrolled_text(self, parent, **kw):
        """主题化 Text + 竖滚动条组合（同上，经典 tk 控件手工上色），
        返回 (外框, 文本框)。"""
        c = self.style.colors

        kw.setdefault("font", FONT)
        kw.setdefault("relief", "flat")
        kw.setdefault("borderwidth", 0)
        kw.setdefault("highlightthickness", 1)

        txt.configure(yscrollcommand=sb.set)

     # ---------- 线程调度 / 基础 ----------

    def _ui(self, fn, *args, **kwargs):
        """把 fn 异步调度到主线程执行（不等待）。"""
        try:
            self.win.after(0, lambda: fn(*args, **kwargs))
        except tk.TclError:
            pass  # 窗口已关闭

    def _ui_sync(self, fn, *args, **kwargs):
        """把 fn 调度到主线程执行并等待结果（后台线程专用）。"""
        if threading.current_thread() is threading.main_thread():
            return fn(*args, **kwargs)
        done = threading.Event()
        box = {}

        def _run():
            try:
                box["ret"] = fn(*args, **kwargs)
            except BaseException as e:
                box["err"] = e
            finally:
                done.set()

        try:
            self.win.after(0, _run)
        except tk.TclError:
            raise RuntimeError("窗口已关闭")
        done.wait()
        if "err" in box:
            raise box["err"]
        return box.get("ret")

    def _set_busy(self, busy):
        self.busy = busy
        state = "disabled" if busy else "normal"
        self.b_run.config(state=state)
        self.b_exec_queue.config(state=state)

    def _check_busy(self):
        if self.busy:
            self.log_print("[提示] 正在执行任务，请等它跑完再操作")
            return True
        return False

    def _open_path(self, target):
        """跨平台打开文件/目录。"""
        target = str(target)
        try:
            if sys.platform == "win32":
                os.startfile(target)
            elif sys.platform == "darwin":
                subprocess.Popen(["open", target])
            else:
                subprocess.Popen(["xdg-open", target])
        except Exception as e:
            self.log_print(f"[提示] 打不开：{target}（{e}）")

    def pick_file(self):
        if self._check_busy():
            return
        p = filedialog.askopenfilename(
            filetypes=[("PDF/图片", "*.pdf *.png *.jpg *.jpeg *.bmp *.tif *.tiff")])
        if p:
            self.e_in.delete(0, "end"); self.e_in.insert(0, p)
            self.try_load_tables()

    def pick_dir(self):
        if self._check_busy():
            return
        p = filedialog.askdirectory()
        if p: self.e_out.delete(0, "end"); self.e_out.insert(0, p)

    def log_print(self, msg):
        """线程安全的日志输出：后台线程里调用会自动转回主线程。"""
        if threading.current_thread() is not threading.main_thread():
            self._ui(self.log_print, msg)
            return
        self.log.insert("end", str(msg) + "\n")
        self.log.see("end")

    # ---------- 队列管理 ----------

    def queue_add(self, kind, payload, label):
        if self._check_busy():
            return
        if kind in ("cmd", "text") and self.order_body is None:
            self.log_print("先拆页出表，再攒指令")
            return
        if kind == "text" and not self.text_rows:
            self.log_print("text.csv 还没就绪，先拆页出表")
            return
        self.queue.append({"kind": kind, "payload": payload})
        self.lb_queue.insert("end", label)
        self.log_print(f"已加入队列（共 {len(self.queue)} 条）：{label}")

    def queue_cmd(self):
        cmd = self.e_cmd.get().strip()
        if not cmd:
            return
        self.queue_add("cmd", cmd, f"[口令] {cmd}")
        self.e_cmd.delete(0, "end")

    def queue_batch_cmd(self):
        if self._check_busy():
            return
        lines = [l.strip() for l in self.t_cmd.get("1.0", "end").splitlines() if l.strip()]
        if not lines:
            self.log_print("批量口令框是空的")
            return
        for l in lines:
            self.queue.append({"kind": "cmd", "payload": l})
            self.lb_queue.insert("end", f"[口令] {l}")
        self.log_print(f"已加入队列 {len(lines)} 条口令（共 {len(self.queue)} 条）")
        self.t_cmd.delete("1.0", "end")

    def queue_reverse(self):
        self.queue_add("reverse", None, "[倒序] 整本倒序")

    def queue_batch_text(self):
        if self._check_busy():
            return
        raw = self.t_batch.get("1.0", "end").strip()
        if not raw:
            self.log_print("批量填字框是空的")
            return
        one = raw.replace("\n", " ")
        self.queue_add("text", one, f"[填字] {one[:40]}{'…' if len(one) > 40 else ''}")
        self.t_batch.delete("1.0", "end")

    def queue_del(self):
        if self._check_busy():
            return
        sel = self.lb_queue.curselection()
        if not sel:
            return
        i = sel[0]
        self.lb_queue.delete(i)
        del self.queue[i]

    def queue_clear(self):
        if self._check_busy():
            return
        self.queue.clear()
        self.lb_queue.delete(0, "end")
        self.log_print("队列已清空")

    def _queue_pop_head(self):
        """主线程调用：移除队首一条（成功的那条），列表同步。"""
        if self.queue:
            self.queue.pop(0)
            self.lb_queue.delete(0)

    def run_queue(self):
        """逐条执行队列：成功一条删一条，失败/体检不过即停 → 重跑只跑剩余的。
        全部执行完且体检通过才出成品。"""
        if self._check_busy():
            return
        pdf = self.input_path()
        if not pdf:
            return
        if not self.queue:
            messagebox.showwarning("提示", "队列是空的，先加几条指令")
            return
        if self.order_body is None:
            self.log_print("先拆页出表")
            return
        self._set_busy(True)
        total = len(self.queue)

        def worker():
            try:
                self.log_print(f"===== 开始逐条执行（共 {total} 条，成功一条删一条）=====")
                # 先把正在编辑、还没落盘的页收进去（批量指令在其后，同页以批量为准）
                self._ui_sync(self._collect_current)
                done, skip = 0, 0
                for i in range(1, total + 1):
                    item = self.queue[0]          # 永远取队首，成功即移除
                    kind, payload = item["kind"], item["payload"]
                    self.log_print(f"--- 第 {i}/{total} 条 ---")
                    if kind == "cmd":
                        buf = io.StringIO()
                        with contextlib.redirect_stdout(buf):
                            changed = order_editor.execute(self.order_body, payload, self.work_dir)
                        out = buf.getvalue().strip()
                        if out:
                            self.log_print("   " + out.replace("\n", "\n   "))
                        if not changed:
                            skip += 1
                            self.log_print("   [没生效] 此条被跳过并移出队列")
                            self._ui_sync(self._queue_pop_head)
                            order_editor.save_table(self.work_dir / "order.csv",
                                                    self.order_header, self.order_body)
                            continue
                        done += 1
                    elif kind == "text":
                        n = self._apply_any_payload(payload)
                        self._ui_sync(self.refresh_text_list)
                        if n:
                            self.log_print(f"   ✔ 批量填字生效 {n} 处")
                            done += 1
                        else:
                            skip += 1
                            self.log_print("   [没生效] 此条被跳过并移出队列")

                    elif kind == "reverse":
                        self._do_reverse_on_body()
                        self.log_print("   ✔ 整本倒序完成")
                        done += 1
                    # ★ 成功一条立即从队列移除——中途停下，剩下的天然是"没执行的"
                    self._ui_sync(self._queue_pop_head)
                    # 每条执行完立即存盘，中途断了也不丢
                    order_editor.save_table(self.work_dir / "order.csv",
                                            self.order_header, self.order_body)
                    self.write_text_csv()
                self._ui_sync(self.load_order)
                self._ui_sync(self.refresh_text_list)
                self.log_print(f"===== 执行完毕：{done} 条生效，{skip} 条没生效并移除 =====")

                # 体检
                errors, warns = [], []
                check(self.work_dir, errors, warns)
                for w in warns:
                    self.log_print(f"[提示] {w}")
                if errors:
                    for e in errors:
                        self.log_print(f"[致命] {e}")
                    self.log_print("表没通过体检，不出成品；修好后重新点执行即可（不会重复已执行的指令）")
                    def _pop():
                        messagebox.showerror(
                            "表格体检没通过",
                            "\n".join(errors[:5]) + ("\n……" if len(errors) > 5 else "")
                            + "\n\n修好两张表后重新点执行即可。")
                    try:
                        self._ui_sync(_pop)
                    except Exception:
                        pass
                    return
                self.log_print("✔ 表格体检通过，开始出成品……")
                final = unique_path(self.final_dir / f"{pdf.stem}_完成.pdf")
                rebuild(self.work_dir, final, Logger(self.log_dir, name="GUI"))
                if fitz is not None:
                    doc = fitz.open(final); n = doc.page_count; doc.close()
                    self.log_print(f"✔ 成品已生成（{n} 页）：{final}")
                else:
                    self.log_print(f"✔ 成品已生成：{final}（未装 PyMuPDF，跳过页数校验）")
                self.log_print("队列已全部执行完毕并出成品")
            except Exception as e:
                self.log_print(f"[致命] {e}")
                self.log_print("已停下；队列里剩的是没执行的条目，修好后重新点执行即可")
            finally:
                self._ui(self._set_busy, False)
        threading.Thread(target=worker, daemon=True).start()

    def load_order(self):
        opath = self.work_dir / "order.csv"
        if not opath.exists():
            return
        self.order_header, self.order_body = order_editor.load_table(opath)

            mark = f"  ←← 图用原第 {rf} 页" if rf else ""
            moved = "  [改过]" if (new_no != old_no or rf) else ""

    def exec_cmd(self):
        if self._check_busy():
            return
        cmd = self.e_cmd.get().strip()
        if not cmd:
            return
        if self.order_body is None:
            self.log_print("先拆页出表，再输口令")
            return
        buf = io.StringIO()
        with contextlib.redirect_stdout(buf):
            changed = order_editor.execute(self.order_body, cmd, self.work_dir)
        text = buf.getvalue().strip()
        if text:
            self.log_print(text)
        if changed:
            order_editor.save_table(self.work_dir / "order.csv",
                                    self.order_header, self.order_body)
            self.load_order()
        self.e_cmd.delete(0, "end")

    def exec_batch_cmd(self):
        if self._check_busy():
            return
        if self.order_body is None:
            self.log_print("先拆页出表")
            return
        lines = [l.strip() for l in self.t_cmd.get("1.0", "end").splitlines() if l.strip()]
        if not lines:
            self.log_print("批量口令框是空的")
            return
        ok, bad = 0, 0
        for line in lines:
            buf = io.StringIO()
            with contextlib.redirect_stdout(buf):
                changed = order_editor.execute(self.order_body, line, self.work_dir)
            out = buf.getvalue().strip()
            self.log_print(f"【{line}】")
            if out:
                self.log_print("   " + out.replace("\n", "\n   "))
            ok += 1 if changed else 0
            bad += 0 if changed else 1
        if ok:
            order_editor.save_table(self.work_dir / "order.csv",
                                    self.order_header, self.order_body)
            self.load_order()
        self.log_print(f"✔ 批量执行完：{ok} 条成功，{bad} 条没生效")
        self.t_cmd.delete("1.0", "end")

    def _do_reverse_on_body(self):
        """在内存 order_body 上整本倒序（不弹窗不存盘，供队列和按钮共用）。"""
        n = len(self.order_body)
        old = [r[1] for r in self.order_body]
        for i, r in enumerate(self.order_body):
            r[1] = old[n - 1 - i]
            r[2] = ""

    # ---------- 插字 ----------

    @staticmethod
    def _norm_row(r):
        """把一行补齐/截断到 TEXT_HEADER 的列数，手工改表缺列也不崩。"""
        r = ["" if c is None else str(c) for c in (r or [])]
        if len(r) < len(TEXT_HEADER):
            r += [""] * (len(TEXT_HEADER) - len(r))
        return r[:len(TEXT_HEADER)]

    def load_text(self):
        tpath = self.work_dir / "text.csv"
        if not tpath.exists():
            self.log_print("text.csv 还没有，先拆页出表")
            return
        with open(tpath, encoding="utf-8-sig") as f:
            rows = list(csv.reader(f))
        self.text_rows = [self._norm_row(r) for r in rows[1:] if r and str(r[0]).strip()]
        self.cur_text_i = None
        self.refresh_text_list()
        try:
            m = json.load(open(self.work_dir / "manifest.json", encoding="utf-8"))
            p0 = next(p for p in m["pages"] if p.get("status") == "已确认")
            self.page_wh = (p0["width_pt"], p0["height_pt"])
            self.lbl_wh.config(text=f"页面约 {self.page_wh[0]:.0f}×{self.page_wh[1]:.0f} 点")
        except Exception:
            self.page_wh = (0, 0)

    def refresh_text_list(self):
        self.lb_text.delete(0, "end")
        for r in self.text_rows:
            preview = (r[1][:12] + "…") if len(r[1]) > 12 else r[1]
            mark = " ✔" if r[1].strip() else ""
            self.lb_text.insert("end", f"第 {r[0]} 页{mark} {preview}")

    def _collect_current(self):
        if self.cur_text_i is None or self.cur_text_i >= len(self.text_rows):
            return False
        r = self.text_rows[self.cur_text_i]
        r[1] = self.e_text.get()
        try:
            r[2] = str(float(self.e_x.get()))

    def _apply_text_payload(self, raw):
        """把 （1）xxx（2）xxx 应用到 text_rows，返回填了几页。"""
        raw = raw.replace("\n", " ")
        parts = re.split(r"[（(](\d+)[)）]", raw)
        if parts[0].strip():
            self.log_print(f"[提示] 开头没页号，忽略：「{parts[0].strip()[:20]}」")
        pairs = list(zip(parts[1::2], parts[2::2]))
        if not pairs:
            self.log_print("[没看懂] 格式示例：（1）封面标题（2）第一章")
            return 0
        filled = []
        for no_s, txt in pairs:
            pno = int(no_s)
            pos = self._where_is(pno)
            if pos and pos != pno:
                self.log_print(f"[提示]（{pno}）指原第{pno}页，"
                               f"它现在排在第 {pos} 位——字跟内容页走，不跟位置")
            # 优先该页空白行，其次该页任意行，最后才新增（页号照抄补零格式）
            r = next((r for r in self.text_rows
                      if r[0].strip().lstrip("0") == str(pno) and not r[1].strip()), None)
            if r is None:
                r = next((r for r in self.text_rows
                          if r[0].strip().lstrip("0") == str(pno)), None)
            if r is None:
                width = len(self.text_rows[0][0]) \
                    if self.text_rows and self.text_rows[0][0].strip().isdigit() else 0
                r = self._norm_row(list(self.text_rows[-1])) if self.text_rows \
                    else self._norm_row([])
                r[0] = str(pno).zfill(width) if width else str(pno)
                r[1] = ""
                r[2] = ""      # ← 补：清掉从上一行抄来的坐标
                r[3] = ""      # ← 补
                r[4] = ""      # ← 补，顺手
                r[5] = ""
                self.text_rows.append(r)
            # 纯照片 PDF 扫不出格子，没坐标就自动给默认位置：
            # 页面左下角（x=50，底边上方 36 点），插完可用点图定位置微调
            if not r[2].strip() or not r[3].strip():
                pw, ph = self._page_size(pno)
                if not r[2].strip():
                    r[2] = "50"
                if not r[3].strip():
                    r[3] = str(round(ph - 36, 1))
                self.log_print(f"[提示]（{pno}）没定坐标，按默认位置"
                               f"（左下角 x=50, y={r[3]}）插入；"
                               f"可用【点图定位置】微调")
            r[1] = txt.strip()
            if not r[4]: r[4] = "12"
            if not r[5]: r[5] = "#000000"
            if not r[6]: r[6] = "left"
            r[7] = "已确认"
            if pno not in filled:
                filled.append(pno)
        self.write_text_csv()
        filled_set = {str(n) for n in filled}
        missing = [r[0] for r in self.text_rows
                   if r[0].strip().lstrip("0") in filled_set and not r[2]]
        self.log_print(f"✔ 填字完成：{len(filled)} 页（{','.join(map(str, filled))}）")
        if missing:
            self.log_print(f"[待办] 这些页还没填坐标：{','.join(missing)}")
        return len(filled)

    def _where_is(self, pno):
        """原第 pno 页现在排在第几位（找不到返回 None）。"""
        for i, r in enumerate(self.order_body):
            try:
                content = r[2].strip() if len(r) > 2 and r[2].strip() \
                          else str(int(r[1]))
            except (ValueError, IndexError):
                continue
            if content.lstrip("0") == str(pno):
                return i + 1
        return None

    def _page_size(self, pno):
        """查原第 pno 页的宽高（点）。查不到就退回全局或 A4。"""
        try:
            m = json.load(open(self.work_dir / "manifest.json", encoding="utf-8"))
            p = next(p for p in m["pages"] if str(p.get("page")) == str(pno))
            return p["width_pt"], p["height_pt"]
        except Exception:
            return self.page_wh if self.page_wh[0] else (595.0, 842.0)

    def _apply_slot_payload(self, raw):
        """把 （3）R2C2=某某公司（3）R3C2=张三 填进格子，返回填了几格。
        同页多格 = 同页多行；同一格重复填 = 覆盖，不涨行；页号格式与表内一致。"""
        spath = self.work_dir / "slots.json"
        slots_map = {}
        if spath.exists():
            try:
                data = json.load(open(spath, encoding="utf-8"))
                for pno, lst in data.items():
                    for s in lst:
                        slots_map[(int(pno), s["行"], s["列"])] = s
            except Exception:
                pass
        if not slots_map:
            self.log_print("[没空位] slots.json 里没有空位数据"
                           "（文件在但为空 = 这个 PDF 没扫出表格，格子口令用不了）")
            return 0
        raw = raw.replace("\n", " ")
        parts = re.split(r"[（(](\d+)[)）]", raw)
        pairs = list(zip(parts[1::2], parts[2::2]))
        ok, bad = 0, 0
        for no_s, body in pairs:
            m = re.match(r"\s*R(\d+)C(\d+)\s*=\s*(.*)", body)
            if not m:
                self.log_print(f"[没看懂] （{no_s}）后面不是 R行C列=文字：「{body[:20]}」")
                bad += 1
                continue
            pno, ri, ci = int(no_s), int(m.group(1)), int(m.group(2))
            txt = m.group(3).strip()
            pos = self._where_is(pno)
            if pos and pos != pno:
                self.log_print(f"[提示]（{pno}）指原第{pno}页，"
                               f"它现在排在第 {pos} 位——字跟内容页走，不跟位置")
            s = slots_map.get((pno, ri, ci))
            if s is None:
                self.log_print(f"[没这格] P{pno}R{ri}C{ci} 不在空位表里，跳过")
                bad += 1
                continue
            x0, y0, x1, y1 = s["矩形"]
            size = s["字号"]
            y = round(y1 - size * 0.30, 1)
            # ① 先找"该页同一格"的行（页号+坐标都相同）→ 直接覆盖，防重复涨行
            def _near(v, target, tol=0.5):
                try:
                    return abs(float(v) - target) < tol
                except (TypeError, ValueError):
                    return False

            r = next((r for r in self.text_rows
                      if r[0].strip().lstrip("0") == str(pno)
                      and _near(r[2], x0) and _near(r[3], y)), None)
            if r is None:
                # ② 该格没填过：优先复用该页的空白行
                r = next((r for r in self.text_rows
                          if r[0].strip().lstrip("0") == str(pno)
                          and not r[1].strip()), None)
                # ③ 真没有才新增，页号照抄表内补零格式（001 而不是 1）
                if r is None:
                    width = len(self.text_rows[0][0]) \
                        if self.text_rows and self.text_rows[0][0].strip().isdigit() else 0
                    r = self._norm_row(list(self.text_rows[-1])) if self.text_rows \
                        else self._norm_row([])
                    r[0] = str(pno).zfill(width) if width else str(pno)
                    r[1] = ""
                    r[2] = ""
                    r[3] = ""
                    r[4] = ""
                    r[5] = ""
                    self.text_rows.append(r)
            r[1], r[2], r[3], r[4] = txt, str(x0), str(y), str(size)
            r[5] = r[5] or "#000000"
            r[6] = "left"
            r[7] = "已确认"
            ok += 1
        self.write_text_csv()
        self.log_print(f"✔ 格子填字完成：{ok} 格生效，{bad} 格没生效")
        return ok

    def _apply_any_payload(self, raw):
        """自动识别：带 R行C列= 的走格子口令，否则走原来的整页填字。"""
        if re.search(r"[（(]\d+[)）]\s*R\d+C\d+\s*=", raw):
            return self._apply_slot_payload(raw)
        return self._apply_text_payload(raw)

    def preview_page(self):
        """弹出这页的预览图：点哪里，x/y 就填哪；红点红字模拟插入效果。"""
        if self._check_busy():
            return
        if self.cur_text_i is None or not self.work_dir:
            messagebox.showwarning("提示", "先在左边选一页")
            return
        self._collect_current()          # 先收当前输入，预览才是最新的
        r = self.text_rows[self.cur_text_i]
        img_path = self.work_dir / f"{r[0]}.png"
        if not img_path.exists():
            self.log_print("这页的图缺失，先拆页出表")
            return
        try:
            from PIL import Image, ImageTk
        except ImportError:
            self.log_print("没装 Pillow，无法预览；pip install pillow")
            return

        c = self.style.colors
        img = Image.open(img_path)
        px_w, px_h = img.size
        pw, ph = self.page_wh
        if not pw:                       # manifest 没读到就粗略按 1px=1pt
            pw, ph = float(px_w), float(px_h)
        sx, sy = pw / px_w, ph / px_h    # 像素→PDF点 的换算比例

        max_w, max_h = 900, 960
        k = min(max_w / px_w, max_h / px_h, 1.0)
        dw, dh = int(px_w * k), int(px_h * k)
        disp = img.resize((dw, dh))

        win = ttkb.Toplevel(self.win)
        win.title(f"第 {r[0]} 页 — 点哪里，字落哪里")
        cv = tk.Canvas(win, width=dw, height=dh,
                       bg=c.inputbg, highlightthickness=1,
                       highlightbackground=c.border)
        cv.pack(padx=10, pady=10)
        cv.image = ImageTk.PhotoImage(disp)   # 持引用，防图片被回收
        cv.create_image(0, 0, anchor="nw", image=cv.image)
        cv.create_text(dw // 2, 16, text="单击图片任意位置 = 字插入那里；改完关窗，点【保存此页】",
                       fill=c.primary, font=FONT)
        # ── 画表格空位框（蓝虚线 + 编号角标）──
        def _load_slots(pno):
            path = self.work_dir / "slots.json"
            if not path.exists():
                return []
            try:
                data = json.load(open(path, encoding="utf-8"))
                return data.get(str(pno), [])
            except Exception:
                return []

        slots = _load_slots(int(r[0]) if str(r[0]).strip().isdigit() else -1)
        for s in slots:
            x0, y0, x1, y1 = (v / sx * k for v in s["矩形"])
            cv.create_rectangle(x0, y0, x1, y1,
                                outline="#1976d2", dash=(4, 3), width=1)
            cv.create_text(x0 + 2, y0 + 1, anchor="nw", text=s["编码"],
                           fill="#1976d2", font=("Microsoft YaHei UI", 7))
        if slots:
            cv.create_text(dw // 2, 34,
                           text=f"本页 {len(slots)} 个空位（蓝虚线框）；点框内=自动吸附",
                           fill="#1976d2", font=FONT)

        marker = []
        def redraw():
            nonlocal marker
            for it in marker:
                cv.delete(it)
            marker = []
            try:
                fx, fy = float(r[2]), float(r[3])
            except ValueError:
                return
            mx, my = fx / sx * k, fy / sy * k
            marker.append(cv.create_oval(mx - 4, my - 4, mx + 4, my + 4,
                                         outline="red", width=2))
            txt = r[1] or "（这里）"
            try:
                fsize = max(int(float(r[4] or 12) * k), 6)
            except ValueError:
                fsize = 10
            marker.append(cv.create_text(mx + 6, my, anchor="w", text=txt[:20],
                                         fill=r[5] or "#000000",
                                         font=("SimSun", fsize)))

        def on_click(e):
            x_pt = e.x / k * sx
            y_pt = e.y / k * sy
            # 先看是不是点在了某个空位框里 → 吸附
            for s in slots:
                bx0, by0, bx1, by1 = s["矩形"]
                if bx0 - 2 <= x_pt <= bx1 + 2 and by0 - 2 <= y_pt <= by1 + 2:
                    r[2] = str(round(bx0, 1))                 # 贴左边框内侧
                    r[3] = str(round(by1 - s["字号"] * 0.30, 1))  # 基线约在底边上方 0.3 字高
                    self.cb_size.set(str(s["字号"]))
                    break
            else:
                # 没点中空位，走原来的手点逻辑
                r[2] = str(round(x_pt, 1))
                r[3] = str(round(y_pt, 1))
            self.e_x.delete(0, "end"); self.e_x.insert(0, r[2])
            self.e_y.delete(0, "end"); self.e_y.insert(0, r[3])
            redraw()
        # ↓↓ 补这两行（注意缩进：8 个空格，和 def on_click 平级）↓↓
        redraw()
        cv.bind("<Button-1>", on_click)

# -*- coding: utf-8 -*-
"""并入另一个 PDF：渲染缩略图 + 扫空位 + 队尾追加 order.csv。
新页从现有最大页号之后续编，统一排在队尾（用拖拽排序调整位置）。
"""

import json
import tempfile
from pathlib import Path

import fitz

import order_editor
import slot_scanner


def merge_pdf(pdf_path: str, work_dir: Path, header, body):
    """把 pdf_path 的所有页并入工作区。body 原地修改并写回 order.csv。"""
    src = fitz.open(pdf_path)
    if len(src) == 0:
        raise ValueError("对方 PDF 一页都没有")

    # ① 续编页号：接在现有最大号后
    used = []
    for r in body:
        try:
            used.append(int(r[1]))
        except (ValueError, IndexError):
            pass
    nxt = (max(used) if used else 0) + 1

    # ② 渲染缩略图 PNG（和拆页同款：三位编号.png）
    from core.common import load_config
    dpi = load_config()["render_dpi"]
    for i, page in enumerate(src):
        pix = page.get_pixmap(dpi=dpi)
        pix.save(work_dir / f"{nxt + i:03d}.png")

    # ②.5 同步 manifest.json：体检靠它算总页数，重建靠它找页面
    # （独立成段，只追加一次；放进渲染循环会导致每渲染一页追加全部新页）
    manifest_path = work_dir / "manifest.json"
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    for i, page in enumerate(src):
        manifest["pages"].append({
            "no": f"{nxt + i:03d}",
            "source": str(pdf_path),
            "page": nxt + i,
            "type": "页面图像",
            "width_pt": round(page.rect.width, 2),
            "height_pt": round(page.rect.height, 2),
            "rotation": page.rotation,
            "dpi": dpi,
            "image": f"{nxt + i:03d}.png",
            "status": "已确认",
        })
    manifest_path.write_text(
        json.dumps(manifest, ensure_ascii=False, indent=2), encoding="utf-8")

    # ③ 扫新 PDF 的空位到临时文件，再并进 slots.json
    slots_path = work_dir / "slots.json"
    with tempfile.TemporaryDirectory() as td:
        tmp = Path(td) / "slots.json"
        slot_scanner.scan_pdf(pdf_path, tmp)
        new_slots = json.loads(tmp.read_text(encoding="utf-8"))
    old_slots = {}
    if slots_path.exists():
        old_slots = json.loads(slots_path.read_text(encoding="utf-8"))
    for i in range(len(src)):
        # 新页在 slots 里的键 = 它在对方 PDF 里的页号（1 起）
        page_slots = []
        for s in new_slots.get(str(i + 1), []):
            s = dict(s)                          # 别改原对象
            s["页"] = nxt + i                    # 页号改成新 PDF 的
            s["编码"] = f"P{nxt + i}R{s['行']}C{s['列']}"   # 编码同步
            page_slots.append(s)
        old_slots[str(nxt + i)] = page_slots
    slots_path.write_text(json.dumps(old_slots, ensure_ascii=False, indent=1),
                          encoding="utf-8")

    # ④ 队尾追加 order 行，写回 order.csv
    added = []
    for i in range(len(src)):
        row = [f"{nxt + i:03d}", str(nxt + i), ""]
        body.append(row)
        added.append(nxt + i)
    order_editor.save_table(work_dir / "order.csv", header, body)
    return added

# -*- coding: utf-8 -*-
"""换页助手：用户说口令（2换3 / 2替3），程序直接改 order.csv，不碰记事本。"""
import sys
import re
import csv
import json
from pathlib import Path

from core.common import ensure_output_dir, unique_path, Logger, FatalError
from core.splitter import split_pdf
from core.rebuilder import rebuild

def load_table(path):
    with open(path, encoding="utf-8-sig") as f:
        rows = list(csv.reader(f))
    return rows[0], rows[1:]          # 表头，数据行

def save_table(path, header, body):
    with open(path, "w", encoding="utf-8-sig", newline="") as f:
        w = csv.writer(f)
        w.writerow(header)
        w.writerows(body)

def find_row(body, new_no):
    for r in body:
        try:
            if int(r[0]) == new_no:
                return r
        except (ValueError, IndexError):
            continue
    return None

# ---------- 显示 ----------

def show(body):
    total = len(body)
    print("\n   当前页序：")
    for r in body:
        new_no, old_no, rf = int(r[0]), int(r[1]), r[2].strip()
        mark = f"  ←← 图用的是原第 {rf} 页" if rf else ""
        moved = "" if new_no == old_no and not rf else "  [改过]"
        print(f"   第 {new_no} 位 ← 原第 {old_no} 页{mark}{moved}")
    print()

# ---------- 口令解析与执行 ----------

def do_swap(body, a, b):
    """a换b：两个位置的页对调。"""
    ra, rb = find_row(body, a), find_row(body, b)
    if ra is None or rb is None:
        print(f"  [没这页] 总共 {len(body)} 页，{a} 或 {b} 超范围了")
        return False
    ra[1], rb[1] = rb[1], ra[1]
    ra[2], rb[2] = rb[2], ra[2]
    print(f"  ✔ 完成：第 {a} 位和第 {b} 位的页已对调")
    return True

def do_replace(body, a, b):
    """a替b：第 a 位的图换成原第 b 页的图（字也跟着）。"""
    total = len(body)
    ra = find_row(body, a)
    if ra is None or not 1 <= b <= total:
        print(f"  [没这页] 总共 {total} 页，{a} 或 {b} 超范围了")
        return False
    ra[2] = str(b)
    print(f"  ✔ 完成：第 {a} 位的图已换成原第 {b} 页的（字也用 {b} 页的）")
    return True

def do_cancel(body, a):
    """取消某位的强行换图。"""
    ra = find_row(body, a)
    if ra is None:
        print(f"  [没这页] {a} 超范围了")
        return False
    ra[2] = ""
    print(f"  ✔ 完成：第 {a} 位的换图已取消，恢复用自己原页的图")
    return True

def do_reset(body):
    for r in body:
        r[1], r[2] = r[0], ""
    print("  ✔ 完成：全部恢复原始页序，换图清空")
    return True
def do_delete(body, a, b=None):
    """删a / 删a-b：删掉位置，后面的页依次顶上来。"""
    nos = set()
    for r in body:
        try:
            nos.add(int(r[0]))
        except ValueError:
            pass
    if b is None:
        targets = [a] if a in nos else []
    else:
        lo, hi = min(a, b), max(a, b)
        targets = [n for n in range(lo, hi + 1) if n in nos]
    if not targets:
        print(f" [没这页] 要删的页不在当前页序里：{a}" if b is None else f" [没这页] {a}-{b} 里没有可删的页")
        return False
    tset = set(targets)
    body[:] = [r for r in body if int(r[0]) not in tset]
    for i, r in enumerate(body, 1):
        r[0] = f"{i:03d}"
    if b is None:
        print(f" ✔ 已删除第 {a} 位，共 {len(body)} 页（后面依次顶上来）")
    else:
        print(f" ✔ 已删除 {len(targets)} 位（{targets[0]}-{targets[-1]}），共 {len(body)} 页")
    return True

def do_blank(body, a):
    """空a：第 a 位原地变空白页（页数不变，old_no 记 0）。"""
    ra = find_row(body, a)
    if ra is None:
        print(f" [没这页] {a} 超范围了")
        return False
    ra[1] = "0"
    ra[2] = ""
    print(f" ✔ 第 {a} 位已改为空白页（页数不变）")
    return True

def do_restore(body, work_dir):
    """恢复：优先从 origin.csv 快照还原（删掉的页也回来）；没快照退回旧逻辑。"""
    if work_dir is not None:
        o = Path(work_dir) / "origin.csv"
        if o.exists():
            _, body0 = load_table(o)
            body[:] = body0
            print(" ✔ 已恢复到原始页序（删除过的页也回来了）")
            return True
        print(" [提示] 没找到快照 origin.csv，只还原页序，删掉的页找不回来")
    return do_reset(body)

def _ensure_origin(work_dir):
    """第一次删/空前，把盘上的 order.csv 存成快照（老任务没有 origin.csv 的兜底）。"""
    if work_dir is None:
        return
    o = Path(work_dir) / "origin.csv"
    src = Path(work_dir) / "order.csv"
    if not o.exists() and src.exists():
        o.write_text(src.read_text(encoding="utf-8-sig"), encoding="utf-8-sig")

def execute(body, cmd, work_dir=None):
    """解析一条口令。返回 True 表示表被改了。"""
    cmd = cmd.replace(" ", "").replace("，", "").replace(",", "")
    if not cmd:
        return False

    # 删a / 删a-b
    m = re.fullmatch(r"删(\d+)(?:-(\d+))?", cmd)
    if m:
        _ensure_origin(work_dir)
        if m.group(2):
            return do_delete(body, int(m.group(1)), int(m.group(2)))
        return do_delete(body, int(m.group(1)))

    # 空a
    m = re.fullmatch(r"空(\d+)", cmd)
    if m:
        _ensure_origin(work_dir)
        return do_blank(body, int(m.group(1)))

    # a复制b（与 a替b 同义）
    m = re.fullmatch(r"(\d+)复制(\d+)", cmd)
    if m:
        return do_replace(body, int(m.group(1)), int(m.group(2)))

    # a换b / a替b / 取消a
    for word, fn in (("换", do_swap), ("替", do_replace), ("取消", do_cancel)):
        if word == "取消":
            m = re.fullmatch(r"取消(\d+)", cmd)
            if m:
                return fn(body, int(m.group(1)))
        else:
            m = re.fullmatch(r"(\d+)" + word + r"(\d+)", cmd)
            if m:
                return fn(body, int(m.group(1)), int(m.group(2)))

    if cmd in ("恢复", "重置", "还原"):
        return do_restore(body, work_dir)
    if cmd in ("看", "显示", "查看"):
        show(body)
        return False

    print(" [没看懂] 口令写法示例：")
    print(" 2换3 → 第2、3位对调")
    print(" 2替5 → 第2位的图换成原第5页的")
    print(" 取消2 → 取消第2位的换图")
    print(" 删3 → 删掉第3位（后面顶上来，成品变薄）")
    print(" 删2-4 → 范围删除")
    print(" 空3 → 第3位变空白页（页数不变）")
    print(" 恢复 → 回到原始页序（删掉的页也回来）")
    print(" 看 → 看当前页序")
    return False

# -*- coding: utf-8 -*-
"""页面拖拽排序：缩略图网格，拖动换位、单击选中、删页/变空白，实时写回 order.csv。
缩略图直接用 工作文件 里拆页生成的 PNG（不重复渲染）。

import tkinter as tk
import ttkbootstrap as ttk
from PIL import Image, ImageTk

CELL_W, CELL_H = 132, 186
TH_W = 116


class PageDragWindow:
    def __init__(self, app):
        self.app = app
        self.work_dir = app.work_dir
        # 从 order_body 读当前页序（内容跟随：有替就用替的图）
        self.pages = []
        for r in app.order_body:
            try:
                old = int(r[1])
                rf = (r[2].strip() if len(r) > 2 else "")
            except (ValueError, IndexError):
                continue
            self.pages.append({"old": old, "rf": rf})

        self.sel = None      # 选中的位置（0 基）
        self.press = None    # (位置, x, y)
        self.drag_i = None
        self.photos = []     # 兼容旧引用，不再用于防回收
        self._cache = {}     # 缩略图缓存：路径 -> PhotoImage / "missing"
        self._cfg_job = None # Configure 防抖句柄
        self.win = tk.Toplevel(app.win)
        self.win.title("页面拖拽排序（拖动换位，改动实时保存）")
        self.win.geometry("1020x720")

        bar = ttk.Frame(self.win)
        bar.pack(fill="x", padx=8, pady=6)

        ttk.Button(bar, text="关 闭", bootstyle="primary",
                   command=self.close).pack(side="right", padx=3)
        ttk.Label(bar, text="拖动缩略图换位置；单击选中（红框）；改动实时写回 order.csv"
                  ).pack(side="left", padx=12)

        self.canvas = tk.Canvas(self.win, bg="#f2f2f2", highlightthickness=0)
        self.canvas.pack(fill="both", expand=True)
        self.canvas.bind("<Button-1>", self.on_press)
        self.canvas.bind("<B1-Motion>", self.on_motion)
        self.canvas.bind("<ButtonRelease-1>", self.on_release)
        self.canvas.bind("<Configure>", self._on_resize)
        self.canvas.bind_all("<MouseWheel>", self.on_wheel)
        self.win.protocol("WM_DELETE_WINDOW", self.close)
        self.redraw()

    # ---------- 布局换算 ----------
    def _on_resize(self, e=None):
        """Configure 防抖：150ms 内的连续触发合并成一次重绘。"""
        if self._cfg_job:
            return
        self._cfg_job = self.win.after(150, self._do_redraw)

    def _do_redraw(self):
        self._cfg_job = None
        try:
            self.redraw()
        except tk.TclError:
            pass          # 窗口已关，忽略

    def _cols(self):
        w = max(self.canvas.winfo_width(), 1)
        return max(1, w // CELL_W)

    def _pos(self, i):
        cols = self._cols()
        return (i % cols) * CELL_W + 6, (i // cols) * CELL_H + 6

    def _index_at(self, x, y):
        cols = self._cols()
        i = (y // CELL_H) * cols + (x // CELL_W)
        return i if 0 <= i < len(self.pages) else None

    def _target(self, x, y):
        """落点 = 插到哪一格的前面/后面（按指针在该格左半还是右半）。"""
        cols = self._cols()
        col, row = x // CELL_W, y // CELL_H
        t = row * cols + col + (1 if (x % CELL_W) > CELL_W // 2 else 0)
        return max(0, min(t, len(self.pages)))

    def _thumb(self, src):
        """src=0 是空白页；文件缺失返回 'missing'。结果进 _cache。"""
        if src in (0, "0"):
            return None
        p = self.work_dir / f"{int(src):03d}.png"
        if not p.exists():
            return "missing"
        key = str(p)
        if key in self._cache:
            return self._cache[key]
        img = Image.open(p)
        k = TH_W / img.width
        img = img.resize((TH_W, max(1, int(img.height * k))))
        ph = ImageTk.PhotoImage(img)
        self._cache[key] = ph
        return ph

    # ---------- 绘制 ----------
    def redraw(self):
        self.canvas.delete("all")
        for i, p in enumerate(self.pages):
            x, y = self._pos(i)
            ph = self._thumb(p["rf"] if p["rf"] else p["old"])
            if ph is None:
                self.canvas.create_rectangle(x, y, x + TH_W, y + 150,
                                             fill="#d9d9d9", outline="#aaaaaa")
                self.canvas.create_text(x + TH_W // 2, y + 75, text="空白页",
                                        fill="#888888")
            elif ph == "missing":
                self.canvas.create_rectangle(x, y, x + TH_W, y + 150,
                                             fill="#f5d7d7", outline="#cc8888")
                self.canvas.create_text(x + TH_W // 2, y + 75,
                                        text=f"缺图(原{p['old']}页)",
                                        fill="#aa3333", font=("", 8))
            else:
                self.canvas.create_image(x, y, anchor="nw", image=ph)
            outline = "#e53935" if self.sel == i else "#bbbbbb"
            self.canvas.create_rectangle(x, y, x + TH_W, y + 152,
                                         outline=outline, width=2)
            mark = f" ←原{p['rf']}页" if p["rf"] else ""
            self.canvas.create_text(x + 2, y + 155, anchor="nw",
                                    text=f"第{i+1}位 ← 原{p['old']}页{mark}",
                                    font=("Microsoft YaHei UI", 8))
        cols = self._cols()
        rows = (len(self.pages) + cols - 1) // cols + 1
        self.canvas.configure(scrollregion=(0, 0, cols * CELL_W, rows * CELL_H))

    # ---------- 拖拽 ----------
    def on_press(self, e):
        self.press = (self._index_at(e.x, e.y), e.x, e.y)
        self.drag_i = None
        self.copying = bool(e.state & 0x0004)   # Ctrl 按住 = 复制模式

    def on_motion(self, e):
        if self.press is None:
            return
        i0, x0, y0 = self.press
        if abs(e.x - x0) + abs(e.y - y0) < 8:
            return
        self.drag_i = i0
        self.canvas.delete("ind")
        t = self._target(e.x, e.y)
        px, py = (self._pos(len(self.pages) - 1)[0] + CELL_W - 8,
                  self._pos(len(self.pages) - 1)[1]) if t >= len(self.pages) \
                 else self._pos(t)
        self.canvas.create_line(px, py + 4, px, py + CELL_H - 8,
                                fill="#e53935", width=3, tags="ind")

    def on_release(self, e):
        self.canvas.delete("ind")
        i0 = self.press[0] if self.press else None
        self.press = None
        if self.drag_i is not None:
            if self.copying:
                tgt = self._index_at(e.x, e.y)
                self.drag_i = None
                if tgt is not None and tgt != i0:
                    src = self.pages[i0]
                    content = src["rf"] if src["rf"] else src["old"]
                    if content in (0, "0"):
                        self.app.log_print("[拖拽排序] 空白页没法复制")
                    else:
                        self.pages[tgt]["rf"] = str(content)
                        self.sel = None
                        self.save(f"第{tgt+1}位已复制原第{content}页（原内容被覆盖）")
                        self.redraw()
                return
            t = self._target(e.x, e.y)
            if t > i0:
                t -= 1
            if t != i0:
                self.pages.insert(t, self.pages.pop(i0))
                self.sel = None
                self.save("页序已调整")
            self.drag_i = None
            self.redraw()
            return
        if i0 is not None:
            self.sel = None if self.sel == i0 else i0
            self.redraw()

    def on_wheel(self, e):
        self.canvas.yview_scroll(-int(e.delta / 120), "units")

    # ---------- 删页 / 空白 ----------
    def del_sel(self):
        if self.sel is None:
            return
        del self.pages[self.sel]
        self.sel = None
        self.save("已删除 1 页")
        self.redraw()

    def blank_sel(self):
        if self.sel is None:
            return
        self.pages[self.sel]["old"] = 0
        self.pages[self.sel]["rf"] = ""
        self.save("已改为空白页")
        self.redraw()

    # ---------- 保存：写回 order.csv（new_no 自动 1..N 连续） ----------
    def save(self, msg):
        import order_editor
        body = [[f"{i:03d}", str(p["old"]), p["rf"]]
                for i, p in enumerate(self.pages, 1)]
        self.app.order_body[:] = body
        order_editor.save_table(self.work_dir / "order.csv",
                                self.app.order_header, self.app.order_body)
        self.app.load_order()
        self.app.log_print(f"[拖拽排序] {msg}（现在共 {len(body)} 位）")

    def close(self):
        # 把全局滚轮还回主窗口
        try:
            c = getattr(self.app.sf, "canvas", None)
            if c is not None:
                c.bind_all("<MouseWheel>",
                           lambda e: c.yview_scroll(-int(e.delta / 120), "units"))
        except Exception:
            pass
        self.win.destroy()


def open_window(app):
    PageDragWindow(app)

# -*- coding: utf-8 -*-
"""slot_scanner.py —— 表格空位扫描器 v0.3

两套扫描，各走各的入口：
  scan_pdf       文字/矢量找表（拆页默认用，快）
  scan_pdf_image 图片找线（纯照片扫描件专用，独立按钮触发，不自动跑）
结果统一存 slots.json，编码格式一致（P页R行C列）。
"""
import json
from pathlib import Path

try:
    import pymupdf as fitz      # 新版
except ImportError:
    import fitz                 # 老版兜底

# ── 字号匹配（文字页专用）──────────────────────────────────
def _页内字号分布(page):
    """统计一页里每个字号出现了多少字，返回 [(字号, 字数), ...] 从多到少。"""
    计数 = {}
    d = page.get_text("dict")
    for block in d["blocks"]:
        for line in block.get("lines", []):
            for span in line["spans"]:
                txt = span["text"].strip()
                if txt:
                    n = 计数.get(round(span["size"], 1), 0) + len(txt)
                    计数[round(span["size"], 1)] = n
    return sorted(计数.items(), key=lambda x: -x[1])

def _找同行字号(page, cell_rect):
    """找空位同一水平线上、位于它左侧最近的文字的字号。
    找不到返回 None，让调用方用本页最常用字号兜底。"""
    d = page.get_text("dict")
    候选 = []   # (左边距离, 字号)
    for block in d["blocks"]:
        for line in block.get("lines", []):
            for span in line["spans"]:
                txt = span["text"].strip()
                if not txt:
                    continue
                r = fitz.Rect(span["bbox"])
                if r.y0 < cell_rect.y1 and r.y1 > cell_rect.y0 \
                        and r.x1 <= cell_rect.x0 + 2:
                    候选.append((cell_rect.x0 - r.x1, span["size"]))
    if 候选:
        候选.sort()
        return round(候选[0][1], 1)
    return None

# ── 第一段：文字/矢量找表 ─────────────────────────────────
def scan_page(page, 页号):
    """扫一页（文字/矢量方式），返回空位清单。没有表格返回空列表。"""
    try:
        tables = page.find_tables().tables
    except Exception:
        return []
    常用字号 = _页内字号分布(page)
    兜底字号 = 常用字号[0][0] if 常用字号 else 12.0
    slots = []
    for t in tables:
        data = t.extract()
        for ri, row in enumerate(t.rows, 1):
            for ci, cell_rect in enumerate(row.cells, 1):
                if cell_rect is None:
                    continue
                if ri > len(data) or ci > len(data[ri - 1]):
                    continue
                内容 = (data[ri - 1][ci - 1] or "").strip()
                if 内容:
                    continue
                rect = fitz.Rect(cell_rect)
                字号 = _找同行字号(page, rect) or 兜底字号
                内缩 = fitz.Rect(rect.x0 + 2, rect.y0 + 2,
                                 rect.x1 - 2, rect.y1 - 2)
                slots.append({
                    "编码": f"P{页号}R{ri}C{ci}",
                    "页": 页号, "行": ri, "列": ci,
                    "矩形": [round(v, 1) for v in 内缩],
                    "字号": 字号,
                })
    return slots

# ── 第二段：图片找线（纯照片件专用，独立入口）───────────────
_暗阈值 = 128          # 灰度低于此算"墨"
_最小格高px = 10       # 比这还矮的格子不认（防噪点）
_最小格宽px = 24

def _渲染灰度(page, dpi):
    """渲染成灰度图。返回 (宽, 高, 像素bytes, 缩放系数)。"""
    zoom = dpi / 72.0
    pix = page.get_pixmap(matrix=fitz.Matrix(zoom, zoom),
                          colorspace=fitz.csGRAY)
    return pix.width, pix.height, pix.samples, zoom

def _runs(mask):
    """一维 bool 数组里的连续段，返回 [(起点, 终点), ...]。numpy 向量化。"""
    import numpy as np
    padded = np.concatenate(([0], mask.astype(np.int8), [0]))
    d = np.diff(padded)
    return list(zip(np.flatnonzero(d == 1).tolist(),
                    np.flatnonzero(d == -1).tolist()))

def _聚合成线(段, 容差=3):
    """段=[(主坐标, 起点, 终点)]。主坐标接近的段并成一条线，
    返回 [(主坐标均值, 最小起点, 最大终点), ...]。"""
    if not 段:
        return []
    段 = sorted(段)
    线, cur = [], [段[0]]
    for s in 段[1:]:
        if s[0] - cur[-1][0] <= 容差:
            cur.append(s)
        else:
            线.append((round(sum(c[0] for c in cur) / len(cur), 1),
                       min(c[1] for c in cur), max(c[2] for c in cur)))
            cur = [s]
    线.append((round(sum(c[0] for c in cur) / len(cur), 1),
               min(c[1] for c in cur), max(c[2] for c in cur)))
    return 线

def scan_page_image(page, 页号, dpi=110):
    """图片方式扫一页：找横竖框线 → 交叉成格 → 内部无墨的算空位。
    优先 numpy（快）；没装则走慢速兜底（先 pip install numpy 更好）。"""
    try:
        import numpy as np
    except ImportError:
        return _scan_page_image_slow(page, 页号, dpi=80)

    w, h, samples, zoom = _渲染灰度(page, dpi)
    dark = (np.frombuffer(samples, dtype=np.uint8)
            .reshape(h, w) < _暗阈值)
    最小横 = max(30, int(w * 0.10))   # 横线至少占页宽 10%
    最小纵 = max(30, int(h * 0.05))   # 竖线至少占页高 5%

    hseg, vseg = [], []
    for y in range(h):
        row = dark[y]
        if row.sum() < 最小横:
            continue
        for x0, x1 in _runs(row):
            if x1 - x0 >= 最小横:
                hseg.append((y, x0, x1))
    for x in range(w):
        col = dark[:, x]
        if col.sum() < 最小纵:
            continue
        for y0, y1 in _runs(col):
            if y1 - y0 >= 最小纵:
                vseg.append((x, y0, y1))

    h线 = _聚合成线(hseg)
    v线 = _聚合成线(vseg)
    if len(h线) < 2 or len(v线) < 2:
        return []

    # 用最长横线定表格左右界，只留横跨/纵贯表格区 60% 以上的线
    最长 = max(h线, key=lambda L: L[2] - L[1])
    X0, X1 = 最长[1], 最长[2]
    Y0, Y1 = min(L[0] for L in h线), max(L[0] for L in h线)
    h线 = [L for L in h线 if min(L[2], X1) - max(L[1], X0) >= 0.6 * (X1 - X0)]
    v线 = [L for L in v线 if min(L[2], Y1) - max(L[1], Y0) >= 0.6 * (Y1 - Y0)]
    if len(h线) < 2 or len(v线) < 2:
        return []

    ys = sorted(set(L[0] for L in h线))
    xs = sorted(set(L[0] for L in v线))

    slots = []
    for ri in range(len(ys) - 1):
        for ci in range(len(xs) - 1):
            py0, py1, px0, px1 = ys[ri], ys[ri + 1], xs[ci], xs[ci + 1]
            if py1 - py0 < _最小格高px or px1 - px0 < _最小格宽px:
                continue
            inner = dark[int(py0) + 2:int(py1) - 2, int(px0) + 2:int(px1) - 2]
            if inner.size and inner.mean() >= 0.02:
                continue                    # 内部有墨 = 有字，不是空位
            字号 = max(9.0, min(14.0, round((py1 - py0) / zoom * 0.55, 1)))
            slots.append({
                "编码": f"P{页号}R{ri + 1}C{ci + 1}",
                "页": 页号, "行": ri + 1, "列": ci + 1,
                "矩形": [round(px0 / zoom, 1), round(py0 / zoom, 1),
                         round(px1 / zoom, 1), round(py1 / zoom, 1)],
                "字号": 字号,
            })
    return slots

def _scan_page_image_slow(page, 页号, dpi=80):
    """没装 numpy 的兜底：纯 Python 逐像素，慢（约 10 秒/页），结果一致。"""
    w, h, data, zoom = _渲染灰度(page, dpi)
    最小横 = max(30, int(w * 0.10))
    最小纵 = max(30, int(h * 0.05))

    def 横段():
        out = []
        for y in range(h):
            base = y * w; x = 0
            while x < w:
                if data[base + x] < _暗阈值:
                    x0 = x
                    while x < w and data[base + x] < _暗阈值: x += 1
                    if x - x0 >= 最小横: out.append((y, x0, x))
                else:
                    x += 1
        return out

    def 竖段():
        out = []
        for x in range(w):
            y = 0
            while y < h:
                if data[y * w + x] < _暗阈值:
                    y0 = y
                    while y < h and data[y * w + x] < _暗阈值: y += 1
                    if y - y0 >= 最小纵: out.append((x, y0, y))
                else:
                    y += 1
        return out

    def 内部墨占比(y0, y1, x0, x1):
        总 = 暗 = 0
        for yy in range(int(y0) + 2, int(y1) - 2):
            base = yy * w
            for xx in range(int(x0) + 2, int(x1) - 2):
                总 += 1
                if data[base + xx] < _暗阈值: 暗 += 1
        return (暗 / 总) if 总 else 1.0

    h线 = _聚合成线(横段()); v线 = _聚合成线(竖段())
    if len(h线) < 2 or len(v线) < 2:
        return []
    最长 = max(h线, key=lambda L: L[2] - L[1])
    X0, X1 = 最长[1], 最长[2]
    Y0, Y1 = min(L[0] for L in h线), max(L[0] for L in h线)
    h线 = [L for L in h线 if min(L[2], X1) - max(L[1], X0) >= 0.6 * (X1 - X0)]
    v线 = [L for L in v线 if min(L[2], Y1) - max(L[1], Y0) >= 0.6 * (Y1 - Y0)]
    if len(h线) < 2 or len(v线) < 2:
        return []
    ys = sorted(set(L[0] for L in h线)); xs = sorted(set(L[0] for L in v线))
    slots = []
    for ri in range(len(ys) - 1):
        for ci in range(len(xs) - 1):
            py0, py1, px0, px1 = ys[ri], ys[ri + 1], xs[ci], xs[ci + 1]
            if py1 - py0 < _最小格高px or px1 - px0 < _最小格宽px:
                continue
            if 内部墨占比(py0, py1, px0, px1) >= 0.02:
                continue
            字号 = max(9.0, min(14.0, round((py1 - py0) / zoom * 0.55, 1)))
            slots.append({
                "编码": f"P{页号}R{ri + 1}C{ci + 1}",
                "页": 页号, "行": ri + 1, "列": ci + 1,
                "矩形": [round(px0 / zoom, 1), round(py0 / zoom, 1),
                         round(px1 / zoom, 1), round(py1 / zoom, 1)],
                "字号": 字号,
            })
    return slots

def scan_pdf_image(pdf路径, json路径=None, 报告=print):
    """【独立入口】图片找线扫整个 PDF，结果并进 slots.json。
    界限：某页已有空位（文字版扫出来的）→ 跳过不覆盖；
    只往没货的页里填图片扫出的空位。返回 {页号: 空位数}。"""
    pdf路径 = Path(pdf路径)
    json路径 = Path(json路径) if json路径 else pdf路径.with_suffix(".slots.json")
    旧 = {}
    if json路径.exists():
        try:
            旧 = json.load(open(json路径, encoding="utf-8"))
        except Exception:
            旧 = {}
    doc = fitz.open(pdf路径)
    新增 = {}
    for pno in range(1, doc.page_count + 1):
        if 旧.get(str(pno)):
            报告(f"页 {pno}：已有空位（文字版扫的），跳过不覆盖")
            continue
        slots = scan_page_image(doc[pno - 1], pno)
        if slots:
            旧[str(pno)] = slots
            新增[pno] = len(slots)
            报告(f"页 {pno}：图片找线扫出 {len(slots)} 个空位")
        else:
            报告(f"页 {pno}：没找到带框线的表格（无框线排版扫不出，用点图定位置）")
    doc.close()
    with open(json路径, "w", encoding="utf-8") as f:
        json.dump(旧, f, ensure_ascii=False, indent=1)
    return 新增

# ── 文字/矢量扫描整个 PDF + 存 json ───────────────────────
def scan_pdf(pdf路径, json路径=None, 覆盖=False):
    """扫一个 PDF 的所有页，返回 {页号: [空位...]}，并存 json。
    覆盖=False（默认）：已有 slots.json 时不覆盖。"""
    pdf路径 = Path(pdf路径)
    if json路径 is None:
        json路径 = pdf路径.with_suffix(".slots.json")
    json路径 = Path(json路径)
    if json路径.exists() and not 覆盖:
        print(f"[slot_scanner] {json路径.name} 已存在，不覆盖（要重扫就传 覆盖=True）")
        with open(json路径, "r", encoding="utf-8") as f:
            return json.load(f)

    doc = fitz.open(pdf路径)
    全部 = {}
    for pno in range(1, doc.page_count + 1):
        page = doc[pno - 1]
        slots = scan_page(page, pno)
        if slots:
            全部[str(pno)] = slots
    doc.close()
    with open(json路径, "w", encoding="utf-8") as f:
        json.dump(全部, f, ensure_ascii=False, indent=1)
    return 全部

# -*- coding: utf-8 -*-
"""贴图：把 PNG/JPG 盖到指定页上，位置大小用页面比例记录。
sticker.csv 列：页号, 图片路径, x, y, w（x/y=左上角比例，w=图宽占页宽比例）

import csv
import tempfile
from pathlib import Path

import fitz

HDR = ["页号", "图片", "x", "y", "w"]


def load_stickers(work_dir: Path):
    p = work_dir / "sticker.csv"
    if not p.exists():
        return list(HDR), []
    with open(p, encoding="utf-8-sig", newline="") as f:
        rows = list(csv.reader(f))
    return rows[0], [r for r in rows[1:] if r and r[0].strip()]


def save_stickers(work_dir: Path, header, body):
    with open(work_dir / "sticker.csv", "w", encoding="utf-8-sig",
              newline="") as f:
        w = csv.writer(f)
        w.writerow(header)
        w.writerows(body)


def add_sticker(work_dir: Path, pno: int, img: str):
    """新增一张贴图：默认居中、宽 30%。返回新行。"""
    header, body = load_stickers(work_dir)
    row = [str(pno), str(img), "0.35", "0.35", "0.3"]
    body.append(row)
    save_stickers(work_dir, header, body)
    return row


def paste_on_page(page, rows):
    """rebuilder 调用：把某页的所有贴图盖到 PDF 页上。"""
    pw, ph = page.rect.width, page.rect.height
    for r in rows:
        try:
            img, x, y, w = r[1], float(r[2]), float(r[3]), float(r[4])
        except (ValueError, IndexError):
            continue
        if not (0 <= x < 1 and 0 <= y < 1 and 0.02 <= w <= 2):
            continue
        try:
            pix = fitz.Pixmap(img)
        except RuntimeError:
            continue  # 图片文件没了/坏了，跳过不炸
        aspect = pix.height / pix.width
        rect = fitz.Rect(x * pw, y * ph,
                         x * pw + w * pw, y * ph + w * pw * aspect)
        page.insert_image(rect, filename=img)

class StickerEditor:

    def _page_rect(self):
        cw = max(self.canvas.winfo_width() - 20, 100)
        ch = max(self.canvas.winfo_height() - 20, 100)
        iw = self.tk_img.width()
        ih = self.tk_img.height()
        s = min(cw / iw, ch / ih, 1.0) * self.zoom
        w, h = iw * s, ih * s
        return 10 + self.offx, 10 + self.offy, w, h, s

    def redraw(self):
        self.canvas.delete("all")
        if self._bg_img is None:
            self._bg_img = self.tk.PhotoImage(file=str(self.png))
        self.tk_img = self._bg_img            # _page_rect 要读它的宽高
        x0, y0, w, h, s = self._page_rect()
        self.canvas.create_image(x0, y0, anchor="nw", image=self.tk_img)
        self._imgs.clear()
        for i, r in enumerate(self.rows):
            fx, fy, fw = float(r[2]), float(r[3]), float(r[4])
            ix, iy = x0 + fx * w, y0 + fy * h
            iw = fw * w
            key = (r[1], int(iw))
            img = self._thumb_cache.get(key)
            if img is None:
                img = self._thumb(r[1], int(iw))
                if img is not None:
                    self._thumb_cache[key] = img
            if img is None:
                self.canvas.create_rectangle(ix, iy, ix + iw, iy + 40,
                                             outline="red")
                continue
            self.canvas.create_image(ix, iy, anchor="nw", image=img)
            self._imgs[i] = img
            if i == self.sel:
                self.canvas.create_rectangle(ix - 2, iy - 2,
                                             ix + iw + 2, iy + img.height() + 2,
                                             outline="#e33", width=2)
        self._geo = (x0, y0, w, h)

    def _thumb(self, path, width):
        """把任意图片转成 PNG 临时文件再上画布（PhotoImage 不认 JPG）。"""
        try:
            pix = fitz.Pixmap(path)
            if pix.alpha:
                pix = fitz.Pixmap(fitz.csRGB, pix)
            if width < 4:
                width = 4
            f = max(width / pix.width, 0.01)
            pix = fitz.Pixmap(pix, int(pix.width * f), int(pix.height * f))
            tmp = Path(tempfile.gettempdir()) / "stk_tmp.png"
            pix.save(tmp)
            return self.tk.PhotoImage(file=str(tmp))
        except RuntimeError:
            return None

    def _hit(self, mx, my):
        x0, y0, w, h = self._geo
        for i, r in enumerate(self.rows):
            fx, fy, fw = float(r[2]), float(r[3]), float(r[4])
            ix, iy = x0 + fx * w, y0 + fy * h
            img = self._imgs.get(i)
            ih = img.height() if img else 40
            if ix <= mx <= ix + fw * w and iy <= my <= iy + ih:
                return i
        return None

    def on_press(self, e):
        self.sel = self._hit(e.x, e.y)
        self.pan = None if self.sel is not None else (e.x, e.y)
        self.drag = (e.x, e.y)
        self.redraw()

    def on_motion(self, e):
        if self.pan is not None:                 # 空白处拖动 = 平移整页
            self.offx += e.x - self.pan[0]
            self.offy += e.y - self.pan[1]
            self.pan = (e.x, e.y)
            self.redraw()
            return
        if self.sel is None or self.drag is None:
            return
        x0, y0, w, h = self._geo
        dx = (e.x - self.drag[0]) / w
        dy = (e.y - self.drag[1]) / h
        r = self.rows[self.sel]
        r[2] = f"{min(max(float(r[2]) + dx, 0), 0.98):.4f}"
        r[3] = f"{min(max(float(r[3]) + dy, 0), 0.98):.4f}"
        self.drag = (e.x, e.y)
        self.redraw()

    def on_release(self, e):
        self.pan = None
        self.drag = None
        self.save()

    def on_wheel(self, e):
        if self.sel is None or (e.state & 0x0004):   # 没选中 或 Ctrl按住 = 缩放视图
            step = 1.1 if e.delta > 0 else 1 / 1.1
            self.zoom = min(max(self.zoom * step, 0.3), 5.0)
            self.redraw()
            return
        step = 0.03 if e.delta > 0 else -0.03        # 选中贴图 = 缩放贴图本身
        r = self.rows[self.sel]
        r[4] = f"{min(max(float(r[4]) + step, 0.05), 1.5):.4f}"
        self.redraw()
        self.save()

    def save(self):
        all_header, all_rows = load_stickers(self.work)
        keep = [r for r in all_rows if r[0].lstrip("0") != str(self.pno)]
        save_stickers(self.work, all_header, keep + self.rows)

    def close(self):
        self.save()
        self.win.destroy()

def detect_indent(text):
    """识别缩进模式。'不规则'只在块缩进里判，
    续行对齐（括号内跨行）不计入，避免误报。"""
    lines = text.split("\n")
    has_tab = has_space = False
    block_widths = []
    prev = ""
    for l in lines:
        if not l.strip() or l.lstrip().startswith("#"):
            continue
        i = 0
        while i < len(l) and l[i] in (" ", "\t"):
            i += 1
        head = l[:i]
        if "\t" in head and " " in head:
            return "混合(同行tab+空格)⚠"
        if "\t" in head:
            has_tab = True
        elif i > 0:
            has_space = True
            # 只统计"块缩进"：上一行以 : 结尾（且不在括号里）
            if prev.rstrip().endswith(":") and i not in block_widths:
                block_widths.append(i)
        prev = l
    if has_tab and has_space:
        return "混合(tab行+空格行)⚠"
    if has_tab:
        return "tab"
    if has_space:
        if not block_widths:
            return "空格"
        unit = reduce(gcd, block_widths)
        if unit == 1:
            return "不规则空格⚠"
        return f"{unit}空格"
    return "无缩进"


def _expand_lead_tabs(line, w):
    """只展开行首连续的 tab；字符串字面量里的 tab 不动。"""
    i = 0
    while i < len(line) and line[i] == "\t":
        i += 1
    return " " * (i * w) + line[i:]


def normalize(text):
    """行尾统一 \\n；行首 tab 展开；去行尾空白（保留行首缩进）。"""
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    lines = [_expand_lead_tabs(l, TAB_WIDTH).rstrip()
             for l in text.split("\n")]
    return "\n".join(lines)


parts, total, syn_bad = [], 0, 0
for f in sorted(root.rglob("*")):
    if not f.is_file():
        continue
    if any(part in SKIP_DIRS for part in f.relative_to(root).parts[:-1]):
        continue
    if f.suffix.lower() not in INCLUDE:
        continue
    if f == out_path:
        continue

    try:
        text = f.read_text(encoding="utf-8")
        enc = "utf-8"
    except UnicodeDecodeError:
        if f.suffix.lower() == ".py":
            print(f"拒收（.py 不是 UTF-8，防编码暗雷）：{f.relative_to(root)}")
            continue
        text = f.read_text(encoding="gbk", errors="replace")
        enc = "gbk(有损)"
    except Exception as e:
        print(f"跳过（读不了）：{f.relative_to(root)}  {e}")
        continue

    if len(text.encode("utf-8")) > MAX_KB * 1024:
        print(f"跳过（太大）：{f.relative_to(root)}")
        continue

    indent = detect_indent(text)
    text = normalize(text)

    syn = ""
    if f.suffix.lower() == ".py":
        try:
            compile(text, str(f), "exec")
            syn = ", 语法✔"
        except SyntaxError as e:
            syn = f", ⚠语法错：第{e.lineno}行 {e.msg}"
            syn_bad += 1
        except ValueError:
            # 典型：含 \x00（二进制污染 / 拷贝截断）
            syn = ", ⚠文件有非法字符（疑似二进制污染），拒发"
            syn_bad += 1
        except Exception as e:
            # 兜底：不让任何单文件异常把整个打包带走
            syn = f", ⚠compile 未知异常：{type(e).__name__}: {e}"
            syn_bad += 1

    n_lines = text.count("\n") + 1
    rel = f.relative_to(root)
    size = f.stat().st_size
    head = (f"\n{'='*60}\n"
            f"【文件】{rel}  ({size} 字节, {n_lines} 行, "
            f"编码={enc}, 缩进={indent}{syn})\n"
            f"来源: {f.resolve()}\n"
            f"{'='*60}\n")
    parts.append(head + text)
    total += 1
    print(f"已收录：{rel}")

out_path.write_text(PREAMBLE + "".join(parts), encoding="utf-8")
print(f"\n✔ 完成：共收录 {total} 个文件 → {out_path}")
